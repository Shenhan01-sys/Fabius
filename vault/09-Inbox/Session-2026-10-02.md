for i, ln in enumerate(open("universe/bsc-universe.jsonl", encoding="utf-8"), 1):
    r = json.loads(ln); body = {k: v for k, v in r.items() if k != "sha256"}

        g = body.get("gdelt") or {}
        if isinstance(g.get("cols_seen"), dict): g["cols_seen"] = {int(k): v for k, v in g["cols_seen"].items()}
